### 2. 元胞自动机算法

```cpp
bool TileMap::stepWater(int iterations, int& outMinX, int& outMinY, int& outMaxX, int& outMaxY) {
    const int MAX_DOWN = 64;   // 每步最大下落体积
    const int MAX_SIDE = 32;   // 每步最大侧流体积
    
    for (int it = 0; it < iterations; ++it) {
        // 阶段1: 重力下落
        for (int y = H-1; y >= 1; --y) {
            for (int x = 0; x < W; ++x) {
                uint8_t current = m_water[index(x,y)];
                if (current == 0) continue;
                if (get(x, y-1) != TileType::Air) continue;
                
                uint8_t below = m_work[index(x, y-1)];
                int capacity = 255 - below;
                int flow = min(current, capacity, MAX_DOWN);
                
                if (flow > 0) {
                    m_work[index(x,y)]   -= flow;
                    m_work[index(x,y-1)] += flow;
                }
            }
        }
        
        // 阶段2: 水平均衡
        for (int y = 0; y < H; ++y) {
            for (int x = 0; x < W-1; ++x) {
                int levelA = m_work[index(x,y)];
                int levelB = m_work[index(x+1,y)];
                int diff = levelA - levelB;
                
                if (abs(diff) > 1) {
                    int flow = min(abs(diff)/2, MAX_SIDE);
                    if (diff > 0) {
                        m_work[index(x,y)]   -= flow;
                        m_work[index(x+1,y)] += flow;
                    } else {
                        m_work[index(x,y)]   += flow;
                        m_work[index(x+1,y)] -= flow;
                    }
                }
            }
        }
    }
}
```

**算法特点：**
- 两阶段更新：重力优先，然后水平均衡
- 体积限制机制防止瞬间大量流动
- 双缓冲确保计算一致性

### 5. 物理力学模拟

```cpp
// 水中物体的流体力学
if (tilemap.isWater(tx, ty)) {
    // 浮力：抵消重力
    float mass = b2Body_GetMass(debris.body);
    b2Vec2 buoyancy = {-gravity.x * mass * 0.85f, -gravity.y * mass * 0.85f};
    
    // 阻力：与速度相反
    b2Vec2 velocity = b2Body_GetLinearVelocity(debris.body);
    b2Vec2 drag = {-1.2f * velocity.x, -1.2f * velocity.y};
    
    // 流力：跟随水流方向
    float flowX, flowY;
    tilemap.waterFlow(position.x, position.y, flowX, flowY);
    b2Vec2 flowForce = {2.0f * (flowX - velocity.x), 2.0f * (flowY - velocity.y)};
    
    b2Body_ApplyForceToCenter(debris.body, buoyancy + drag + flowForce, true);
}
```

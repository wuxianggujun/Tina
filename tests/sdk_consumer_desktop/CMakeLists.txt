cmake_minimum_required(VERSION 3.25)
project(TinaSdkDesktopConsumer LANGUAGES CXX)

# Conventional names belonging to another user of bgfx must not be hijacked.
add_library(bgfx::bgfx INTERFACE IMPORTED)
add_library(bgfx::bx INTERFACE IMPORTED)
add_library(bgfx::bimg INTERFACE IMPORTED)
find_package(Tina 0.3.0 EXACT CONFIG REQUIRED COMPONENTS Desktop)
include("${CMAKE_CURRENT_LIST_DIR}/../sdk_consumer/VerifyInstalledTargets.cmake")
tina_verify_installed_targets(${Tina_PACKAGE_TARGETS})
tina_verify_single_library_sdk()
tina_verify_installed_shader_include_dirs(TRUE)
foreach(feature IN ITEMS Desktop PlatformGlfw RenderBgfx)
    if(NOT feature IN_LIST Tina_FEATURES)
        message(FATAL_ERROR "Desktop SDK is missing capability ${feature}")
    endif()
endforeach()
foreach(dependency IN ITEMS TinaBgfxRuntime::bgfx TinaBgfxRuntime::bx TinaBgfxRuntime::bimg)
    if(NOT TARGET ${dependency})
        message(FATAL_ERROR "Desktop SDK is missing private dependency ${dependency}")
    endif()
endforeach()
add_executable(tina_sdk_desktop_bootstrap_consumer main.cpp)
target_compile_features(tina_sdk_desktop_bootstrap_consumer PRIVATE cxx_std_23)
target_link_libraries(tina_sdk_desktop_bootstrap_consumer PRIVATE Tina::GameSDK)
set_target_properties(tina_sdk_desktop_bootstrap_consumer PROPERTIES CXX_EXTENSIONS OFF)

cmake_minimum_required(VERSION 3.25)
project(TinaSdkMissingCapability LANGUAGES NONE)
if(NOT TINA_EXPECT_MISSING_COMPONENTS)
    message(FATAL_ERROR "TINA_EXPECT_MISSING_COMPONENTS must name unavailable capabilities")
endif()
find_package(Tina 0.3.0 EXACT CONFIG COMPONENTS ${TINA_EXPECT_MISSING_COMPONENTS})
if(Tina_FOUND)
    message(FATAL_ERROR "The SDK accepted unavailable capabilities: ${TINA_EXPECT_MISSING_COMPONENTS}")
endif()
foreach(feature IN LISTS TINA_EXPECT_MISSING_COMPONENTS)
    if(Tina_${feature}_FOUND)
        message(FATAL_ERROR "Unavailable capability ${feature} is marked found")
    endif()
endforeach()
if(TARGET Tina::GameSDK)
    message(FATAL_ERROR "A failed capability request partially published the SDK target")
endif()
